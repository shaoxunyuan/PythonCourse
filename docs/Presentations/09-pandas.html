流水线：函数化示意

def clean_cases(path):
    df = pd.read_csv(path, na_values=["?", "NA", ""])
    cols = ["encounter_id", "gender", "time_in_hospital", "readmitted"]
    df = df[cols].copy()
    df = df.drop_duplicates(subset=["encounter_id"])
    df["time_in_hospital"] = pd.to_numeric(
        df["time_in_hospital"], errors="coerce")
    df = df.dropna(subset=["time_in_hospital", "gender"])
    df["gender"] = df["gender"].str.strip().str.lower()
    return df.reset_index(drop=True)

代码：缺失报告小工具

def missing_report(df):
    n = len(df)
    s = df.isna().sum()
    out = pd.DataFrame({
        "n_missing": s,
        "rate": (s / n).round(4),
    })
    return out[out["n_missing"] > 0].sort_values(
        "rate", ascending=False
    )

print(missing_report(df))
